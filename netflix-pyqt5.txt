# VIDEO OYNATICI
class VideoOynatici(QWidget):
    def __init__(self, video_dosyasi, baslik, kullanici_id=None, program_id=None, baslangic_sure=0):
        super().__init__()
        self.setWindowTitle(f"Netflix - {baslik}")
        self.setMinimumSize(800, 500)
        self.setStyleSheet("background-color: #000;")
        self.video_dosyasi = os.path.abspath(video_dosyasi)
        self.kullanici_id = kullanici_id
        self.program_id = program_id
        self.baslangic_sure = baslangic_sure
        self.instance = vlc.Instance()
        self.player = self.instance.media_player_new()
        self.setup_ui()

    def setup_ui(self):
        layout = QVBoxLayout()
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)
        self.video_frame = QFrame()
        self.video_frame.setStyleSheet("background: black;")
        self.video_frame.setMinimumHeight(400)
        kontrol_widget = QWidget()
        kontrol_widget.setStyleSheet("background: #111; padding: 8px;")
        kontrol_layout = QHBoxLayout()
        self.oynat_btn = QPushButton("Durdur")
        self.oynat_btn.setStyleSheet("background: #E50914; color: white; padding: 8px 16px; border-radius: 4px; font-weight: bold;")
        self.oynat_btn.clicked.connect(self.oynat_durdur)
        self.ilerleme = QSlider(Qt.Horizontal)
        self.ilerleme.setMaximum(1000)
        self.ilerleme.setStyleSheet("QSlider::handle:horizontal { background: #E50914; width: 12px; }")
        self.ilerleme.sliderMoved.connect(self.pozisyon_degistir)
        self.sure_label = QLabel("00:00 / 00:00")
        self.sure_label.setStyleSheet("color: white; font-size: 12px;")
        ses_label = QLabel
        ses_label.setStyleSheet("color: white;")
        self.ses_slider = QSlider(Qt.Horizontal)
        self.ses_slider.setMaximum(100)
        self.ses_slider.setValue(80)
        self.ses_slider.setMaximumWidth(100)
        self.ses_slider.setStyleSheet("QSlider::handle:horizontal { background: white; width: 12px; }")
        self.ses_slider.valueChanged.connect(self.ses_degistir)
        tamamla_btn = QPushButton(" İzlemeyi Tamamla")
        tamamla_btn.setStyleSheet("background: #28a745; color: white; padding: 8px; border-radius: 4px; font-weight: bold;")
        tamamla_btn.clicked.connect(self.izlemeyi_tamamla)
        kapat_btn = QPushButton(" Kapat")
        kapat_btn.setStyleSheet("background: #333; color: white; padding: 8px; border-radius: 4px;")
        kapat_btn.clicked.connect(self.kapat)
        kontrol_layout.addWidget(self.oynat_btn)
        kontrol_layout.addWidget(self.ilerleme)
        kontrol_layout.addWidget(self.sure_label)
        kontrol_layout.addWidget(ses_label)
        kontrol_layout.addWidget(self.ses_slider)
        kontrol_layout.addWidget(tamamla_btn)
        kontrol_layout.addWidget(kapat_btn)
        kontrol_widget.setLayout(kontrol_layout)
        layout.addWidget(self.video_frame)
        layout.addWidget(kontrol_widget)
        self.setLayout(layout)
        media = self.instance.media_new(self.video_dosyasi)
        self.player.set_media(media)
        self.player.set_hwnd(int(self.video_frame.winId()))
        self.player.audio_set_volume(80)
        self.player.play()
        if self.baslangic_sure > 0:
            QTimer.singleShot(1000, lambda: self.player.set_time(self.baslangic_sure * 1000))
        self.timer = QTimer()
        self.timer.setInterval(500)
        self.timer.timeout.connect(self.sure_guncelle)
        self.timer.start()

    def oynat_durdur(self):
        if self.player.is_playing():
            self.player.pause()
            self.oynat_btn.setText(" Oynat")
        else:
            self.player.play()
            self.oynat_btn.setText(" Durdur")

    def pozisyon_degistir(self, deger):
        self.player.set_position(deger / 1000.0)

    def ses_degistir(self, deger):
        self.player.audio_set_volume(deger)

    def sure_guncelle(self):
        if self.player.get_length() > 0:
            sure = self.player.get_length() // 1000
            pozisyon = self.player.get_time() // 1000
            self.ilerleme.setValue(int(self.player.get_position() * 1000))
            self.sure_label.setText(f"{pozisyon//60:02d}:{pozisyon%60:02d} / {sure//60:02d}:{sure%60:02d}")

    def kalinan_sureyi_kaydet(self):
        if self.kullanici_id and self.program_id:
            sure_sn = self.player.get_time() // 1000
            try:
                conn = get_connection()
                cursor = conn.cursor(buffered=True)
                cursor.execute("UPDATE izleme_gecmisi SET kalinan_sure=%s WHERE kullanici_id=%s AND program_id=%s",
                               (sure_sn, self.kullanici_id, self.program_id))
                conn.commit()
                cursor.close()
                conn.close()
            except Exception as e:
                print(f"Süre kaydetme hatası: {e}")

    def izlemeyi_tamamla(self):
        if self.kullanici_id and self.program_id:
            try:
                self.timer.stop()
                self.player.stop()
                conn = get_connection()
                cursor = conn.cursor(buffered=True)
                cursor.execute("UPDATE izleme_gecmisi SET tamamlandi=1, kalinan_sure=0 WHERE kullanici_id=%s AND program_id=%s",
                               (self.kullanici_id, self.program_id))
                conn.commit()
                cursor.close()
                conn.close()
                QTimer.singleShot(300, self._tamamlandi_mesaj)
            except Exception as e:
                print(f"Tamamlama hatası: {e}")

    def _tamamlandi_mesaj(self):
        QMessageBox.information(self, "Tamamlandı", "İzleme tamamlandı olarak işaretlendi!")
        self.close()

    def kapat(self):
        self.kalinan_sureyi_kaydet()
        self.player.stop()
        self.timer.stop()
        self.close()

    def closeEvent(self, event):
        self.kalinan_sureyi_kaydet()
        self.player.stop()
        self.timer.stop()
        event.accept()

class LoginEkrani(QWidget):

        giris_btn.clicked.connect(self.giris_yap)
        kayit_btn = QPushButton("Hesabın yok mu? Kayıt Ol")
        kayit_btn.setStyleSheet("background: none; color: #aaa; border: none;")
        kayit_btn.clicked.connect(self.kayit_ac)

        self.hata_label.setStyleSheet("color: red;")

        layout.addWidget(self.sifre_input)
        layout.addWidget(giris_btn)
        layout.addWidget(kayit_btn)
        layout.addWidget(self.hata_label)
        self.setLayout(layout)

    def giris_yap(self):
        email = self.email_input.text().strip()
        sifre = self.sifre_input.text().strip()
        if not email or not sifre:
            self.hata_label.setText("Lütfen tüm alanları doldurun!")
            return
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("SELECT kullanici_id, ad, rol_id FROM kullanici WHERE email=%s AND sifre=%s", (email, sifre))
            kullanici = cursor.fetchone()
            cursor.close()
            conn.close()
            if kullanici:
                self.ana = AnaSayfa(kullanici[0], kullanici[1], kullanici[2])
                self.ana.show()
                self.close()
            else:
                self.hata_label.setStyleSheet("color: red;")
                self.hata_label.setText("Email veya şifre hatalı!")
        except Exception as e:
            self.hata_label.setText(f"Hata: {e}")

    def kayit_ac(self):
        self.kayit = KayitEkrani(self)
        self.kayit.show()
        self.hide()

        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("SELECT tur_id, tur_adi FROM tur ORDER BY tur_adi")
            for tur_id, tur_adi in cursor.fetchall():
                item = QListWidgetItem(tur_adi)
                item.setData(Qt.UserRole, tur_id)
                self.tur_listesi.addItem(item)
            cursor.close()
            conn.close()
        except Exception as e:
            print(f"Tür yükleme hatası: {e}")

    def kayit_ol(self):
        ad = self.ad_input.text().strip()
        soyad = self.soyad_input.text().strip()
        email = self.email_input.text().strip()
        sifre = self.sifre_input.text().strip()
        sifre2 = self.sifre2_input.text().strip()
        dogum = self.dogum_input.date()
        dogum_str = dogum.toString("yyyy-MM-dd")
        cinsiyet = self.cinsiyet_input.currentText()
        ulke = self.ulke_input.text().strip()
        secilen_turler = self.tur_listesi.selectedItems()
        if not all([ad, soyad, email, sifre, sifre2, ulke]):
            self.hata_label.setText("Lütfen tüm alanları doldurun!")
            return
        bugun = QDate(2026, 5, 18)
        if dogum > bugun:
            self.hata_label.setText("Doğum tarihi bugünden büyük olamaz!")
            return
        if sifre != sifre2:
            self.hata_label.setText("Şifreler eşleşmiyor!")
            return
        if len(sifre) < 6:
            self.hata_label.setText("Şifre en az 6 karakter olmalı!")
            return
        if len(secilen_turler) != 3:
            self.hata_label.setText("Lütfen tam olarak 3 tür seçin!")
            return
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("SELECT kullanici_id FROM kullanici WHERE email=%s", (email,))
            if cursor.fetchone():
                self.hata_label.setText("Bu email zaten kayıtlı!")
                cursor.close()
                conn.close()
                return
            cursor.execute("INSERT INTO kullanici (ad, soyad, email, sifre, dogum_tarihi, cinsiyet, ulke, rol_id) VALUES (%s,%s,%s,%s,%s,%s,%s,%s)",
                           (ad, soyad, email, sifre, dogum_str, cinsiyet, ulke, 2))
            kullanici_id = cursor.lastrowid
            for item in secilen_turler:
                tur_id = item.data(Qt.UserRole)
                cursor.execute("INSERT INTO kullanici_tur (kullanici_id, tur_id) VALUES (%s,%s)", (kullanici_id, tur_id))
            conn.commit()
            cursor.close()
            conn.close()
            self.hata_label.setStyleSheet("color: green;")
            self.hata_label.setText("Kayıt başarılı! Giriş yapabilirsiniz.")
        except Exception as e:
            self.hata_label.setText(f"Hata: {e}")

    def geri_don(self):
        self.close()
        self.login_ekrani.show()

            query = """
                SELECT p.program_id, p.ad,
                       GROUP_CONCAT(DISTINCT t.tur_adi SEPARATOR ', ') as turler,
                       p.tip, p.yayin_yili,
                       COALESCE(ROUND(AVG(pu.skor), 1), 0) as ort_puan
                FROM program p
                LEFT JOIN program_tur pt ON p.program_id = pt.program_id
                LEFT JOIN tur t ON pt.tur_id = t.tur_id
                LEFT JOIN puan pu ON p.program_id = pu.program_id
                WHERE 1=1
            """
            params = []
            if arama:
                query += " AND p.ad LIKE %s"
                params.append(f"%{arama}%")
            if tip != "Tümü":
                query += " AND p.tip = %s"
                params.append(tip)
            if tur_id:
                query += " AND pt.tur_id = %s"
                params.append(tur_id)
            if yil.strip():
                query += " AND p.yayin_yili = %s"
                params.append(yil.strip())
            query += " GROUP BY p.program_id"
            if min_puan > 0:
                query += " HAVING ort_puan >= %s"
                params.append(min_puan)
            if siralama == "En Çok İzlenen":
                query += " ORDER BY p.izlenme_sayisi DESC"
            elif siralama == "En Yüksek Puanlı":
                query += " ORDER BY ort_puan DESC"
            else:
                query += " ORDER BY p.ad"
            cursor.execute(query, params)
            sonuclar = cursor.fetchall()
            cursor.close()
            conn.close()
            self.tablo.setRowCount(len(sonuclar))
            for i, row in enumerate(sonuclar):
                for j in range(5):
                    val = row[j]
                    item = QTableWidgetItem(str(val) if val else "")
                    item.setTextAlignment(Qt.AlignCenter)
                    self.tablo.setItem(i, j, item)
                puan_val = row[5]
                puan_str = f"{puan_val}/10" if puan_val and float(puan_val) > 0 else "-"
                puan_item = QTableWidgetItem(puan_str)
                puan_item.setTextAlignment(Qt.AlignCenter)
                self.tablo.setItem(i, 5, puan_item)
        except Exception as e:
            print(f"İçerik yükleme hatası: {e}")

            cursor.execute("""SELECT DISTINCT t.tur_id, t.tur_adi FROM favori f
                JOIN program_tur pt ON f.program_id = pt.program_id
                JOIN tur t ON pt.tur_id = t.tur_id
                WHERE f.kullanici_id = %s ORDER BY t.tur_adi""", (self.kullanici_id,))

    def favorileri_yukle(self, tur_id=0):
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            query = """SELECT p.program_id, p.ad, GROUP_CONCAT(DISTINCT t.tur_adi SEPARATOR ', '), p.tip
                FROM favori f JOIN program p ON f.program_id = p.program_id
                LEFT JOIN program_tur pt ON p.program_id = pt.program_id
                LEFT JOIN tur t ON pt.tur_id = t.tur_id
                WHERE f.kullanici_id = %s"""
            params = [self.kullanici_id]
            if tur_id:
                query += " AND pt.tur_id = %s"
                params.append(tur_id)
            query += " GROUP BY p.program_id ORDER BY p.ad"
            cursor.execute(query, params)
            rows = cursor.fetchall()
            cursor.close()
            conn.close()
            self.tablo.setRowCount(len(rows))
            for i, row in enumerate(rows):
                for j, val in enumerate(row):
                    item = QTableWidgetItem(str(val) if val else "")
                    item.setTextAlignment(Qt.AlignCenter)
                    self.tablo.setItem(i, j, item)
            self.sayi_label.setText(f"Toplam: {len(rows)} içerik")
        except Exception as e:
            print(f"Favori yükleme hatası: {e}")

        self.favorileri_yukle(self.tur_combo.currentData() or 0)

    def detay_ac(self):
        secili = self.tablo.currentRow()
        if secili >= 0:
            program_id = int(self.tablo.item(secili, 0).text())
            program_adi = self.tablo.item(secili, 1).text()
            self.detay = DetayEkrani(program_id, program_adi, self.kullanici_id)
            self.detay.show()

    def favoriden_cikar(self):
        secili = self.tablo.currentRow()
        if secili < 0:
            self.sonuc_label.setStyleSheet("color: red;")
            self.sonuc_label.setText("Lütfen bir içerik seçin!")
            return
        program_id = int(self.tablo.item(secili, 0).text())
        program_adi = self.tablo.item(secili, 1).text()
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("DELETE FROM favori WHERE kullanici_id=%s AND program_id=%s", (self.kullanici_id, program_id))
            conn.commit()
            cursor.close()
            conn.close()
            self.sonuc_label.setStyleSheet("color: green;")
            self.sonuc_label.setText(f"'{program_adi}' favorilerden çıkarıldı!")
            self.tur_combo.blockSignals(True)
            self.tur_combo.clear()
            self.tur_combo.addItem("Tüm Türler", 0)
            self.turleri_yukle()
            self.tur_combo.blockSignals(False)
            self.favorileri_yukle()
        except Exception as e:
            self.sonuc_label.setStyleSheet("color: red;")
            self.sonuc_label.setText(f"Hata: {e}")

        for i, row in enumerate(rows):
            sira = QTableWidgetItem(str(i + 1))
            sira.setTextAlignment(Qt.AlignCenter)
            tablo.setItem(i, 0, sira)
            for j, val in enumerate(row[1:]):
                puan_str = f"{val}/10 " if j == 2 and val and float(val) > 0 else str(val) if val else "-"
                item = QTableWidgetItem(puan_str)
                item.setTextAlignment(Qt.AlignCenter)
                tablo.setItem(i, j + 1, item)

    def detay_ac(self, tablo):
        secili = tablo.currentRow()
        if secili >= 0:
            icerik_adi = tablo.item(secili, 1).text()
            try:
                conn = get_connection()
                cursor = conn.cursor(buffered=True)
                cursor.execute("SELECT program_id FROM program WHERE ad = %s", (icerik_adi,))
                row = cursor.fetchone()
                cursor.close()

    def kullanici_sil(self):
        secili = self.kullanici_tablo.currentRow()
        if secili < 0:
            self.kullanici_sonuc.setStyleSheet("color: red;")
            self.kullanici_sonuc.setText("Lütfen bir kullanıcı seçin!")
            return
        kullanici_id = int(self.kullanici_tablo.item(secili, 0).text())
        ad = self.kullanici_tablo.item(secili, 1).text()
        cevap = QMessageBox.question(self, "Emin misiniz?", f"'{ad}' silinecek!", QMessageBox.Yes | QMessageBox.No)
        if cevap == QMessageBox.Yes:
            try:
                conn = get_connection()
                cursor = conn.cursor(buffered=True)
                for tbl in ["favori", "izleme_gecmisi", "puan", "kullanici_tur"]:
                    cursor.execute(f"DELETE FROM {tbl} WHERE kullanici_id=%s", (kullanici_id,))
                cursor.execute("DELETE FROM kullanici WHERE kullanici_id=%s", (kullanici_id,))
                conn.commit()
                cursor.close()
                conn.close()
                self.kullanici_sonuc.setStyleSheet("color: green;")
                self.kullanici_sonuc.setText(f"'{ad}' silindi!")
                self.kullanicilari_yukle()
            except Exception as e:
                self.kullanici_sonuc.setText(f"Hata: {e}")

    def istatistik_yukle(self):
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("""SELECT p.ad, p.tip, ROUND(AVG(pu.skor), 1) as ort_puan, COUNT(pu.puan_id) as puan_sayisi
                FROM program p INNER JOIN puan pu ON p.program_id = pu.program_id
                GROUP BY p.program_id HAVING puan_sayisi > 0
                ORDER BY ort_puan DESC, puan_sayisi DESC LIMIT 20""")
            rows = cursor.fetchall()
            cursor.close()
            conn.close()
            self.istatistik_tablo.setRowCount(len(rows))
            for i, row in enumerate(rows):
                sira = QTableWidgetItem(f"#{i+1}")
                sira.setTextAlignment(Qt.AlignCenter)
                if i == 0: sira.setForeground(QColor('#FFD700'))
                elif i == 1: sira.setForeground(QColor('#C0C0C0'))
                elif i == 2: sira.setForeground(QColor('#CD7F32'))
                self.istatistik_tablo.setItem(i, 0, sira)
                for j, val in enumerate(row):
                    item = QTableWidgetItem(str(val) if val else "")
                    item.setTextAlignment(Qt.AlignCenter)
                    self.istatistik_tablo.setItem(i, j + 1, item)
        except Exception as e:
            print(f"İstatistik yükleme hatası: {e}")

    def raporlari_yukle(self):
        try:
            conn = get_connection()
            cursor = conn.cursor(buffered=True)
            cursor.execute("SELECT COUNT(*) FROM kullanici WHERE rol_id=2")
            self.kullanici_sayi_label.setText(f" Kullanıcı: {cursor.fetchone()[0]}")
            cursor.execute("SELECT COUNT(*) FROM izleme_gecmisi")
            self.izlenme_sayi_label.setText(f" Toplam İzlenme: {cursor.fetchone()[0]}")
            cursor.execute("SELECT COUNT(*) FROM puan")
            self.puan_sayi_label.setText(f" Toplam Puan: {cursor.fetchone()[0]}")
            cursor.execute("SELECT p.ad, p.tip, p.izlenme_sayisi FROM program p ORDER BY p.izlenme_sayisi DESC LIMIT 10")
            self._rapor_tablo_doldur(self.en_cok_izlenen_tablo, cursor.fetchall())
            cursor.execute("""SELECT p.ad, p.tip, ROUND(AVG(pu.skor), 1), COUNT(pu.puan_id)
                FROM program p INNER JOIN puan pu ON p.program_id = pu.program_id
                GROUP BY p.program_id HAVING COUNT(pu.puan_id) > 0
                ORDER BY AVG(pu.skor) DESC LIMIT 10""")
            self._rapor_tablo_doldur(self.en_yuksek_puan_tablo, cursor.fetchall())
            cursor.execute("""SELECT t.tur_adi, COUNT(ig.gecmis_id) as izlenme
                FROM tur t JOIN program_tur pt ON t.tur_id = pt.tur_id
                JOIN izleme_gecmisi ig ON pt.program_id = ig.program_id
                GROUP BY t.tur_id ORDER BY izlenme DESC LIMIT 10""")
            self._rapor_tablo_doldur(self.en_cok_tur_tablo, cursor.fetchall())
            cursor.execute("""SELECT CONCAT(k.ad, ' ', k.soyad), k.email,
                COUNT(DISTINCT ig.program_id), COUNT(DISTINCT f.program_id), COUNT(DISTINCT pu.puan_id)
                FROM kullanici k
                LEFT JOIN izleme_gecmisi ig ON k.kullanici_id = ig.kullanici_id
                LEFT JOIN favori f ON k.kullanici_id = f.kullanici_id
                LEFT JOIN puan pu ON k.kullanici_id = pu.kullanici_id
                WHERE k.rol_id = 2 GROUP BY k.kullanici_id
                ORDER BY COUNT(DISTINCT ig.program_id) DESC LIMIT 10""")
            self._rapor_tablo_doldur(self.en_aktif_tablo, cursor.fetchall())
            cursor.execute("""SELECT p.ad, p.tip, CONCAT(k.ad, ' ', k.soyad),
                DATE_FORMAT(ig.son_izleme, '%d/%m/%Y %H:%i')
                FROM izleme_gecmisi ig
                JOIN program p ON ig.program_id = p.program_id
                JOIN kullanici k ON ig.kullanici_id = k.kullanici_id
                WHERE ig.son_izleme >= DATE_SUB(NOW(), INTERVAL 7 DAY)
                ORDER BY ig.son_izleme DESC""")
            self._rapor_tablo_doldur(self.son_7_gun_tablo, cursor.fetchall())
            cursor.close()
            conn.close()
